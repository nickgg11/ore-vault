# E3-S11 (#245) spike, lifetime 1: base Vault unloaded, deleted, made again as expanded.
run orevault vault create 0000000a-0000-4000-8000-00000000000a base
expect type=orevault:ore_vault$
run orevault vault create 0000000b-0000-4000-8000-00000000000b base
expect type=orevault:ore_vault$
# Generate and edit chunks in A, one of them far out (region r.1.1) and force-loaded, so check 4 has a file to look for.
run execute in orevault:vault_0000000a00004000800000000000000a run forceload add 600 600
expect Marked chunk
run execute in orevault:vault_0000000a00004000800000000000000a run setblock 600 251 600 minecraft:gold_block
expect Changed the block
run execute in orevault:vault_0000000a00004000800000000000000a run orevault break 0 250 0
expect gone=true
log \[orevault-debug\] tick orevault:vault_0000000a00004000800000000000000a server_tick=
log \[orevault-debug\] tick orevault:vault_0000000b00004000800000000000000b server_tick=
run save-all flush
expect Saved the game
# Removal steps 1 to 5, the drain advanced once a tick (check 6).
run orevault vault unload 0000000a-0000-4000-8000-00000000000a
expect unloading
log Unloaded Vault orevault:vault_0000000a00004000800000000000000a of team 0000000a-0000-4000-8000-00000000000a \(spike\): drain \d+ ticks
# The old level wrote its far chunk, so the check after the stop has a file to miss.
run orevault vault folder 0000000a-0000-4000-8000-00000000000a
expect exists=true deleting=false far_region=true
# Check 2: the folder moves and deletes at once, straight after the removal line, with no wait.
run orevault vault delete 0000000a-0000-4000-8000-00000000000a
expect deleted
run orevault vault folder 0000000a-0000-4000-8000-00000000000a
expect exists=false deleting=false far_region=false
# Check 1, after the delete: sixty seconds on, no tick line for A after its removal, B still ticking, nothing thrown.
sleep 60
nolog (?s)Unloaded Vault orevault:vault_0000000a00004000800000000000000a.*tick orevault:vault_0000000a00004000800000000000000a server_tick
log (?s)Unloaded Vault orevault:vault_0000000a00004000800000000000000a.*tick orevault:vault_0000000b00004000800000000000000b server_tick
nolog (?s)Unloading Vault orevault:vault_0000000a00004000800000000000000a.*(/ERROR\]|Exception)
# Check 4, first half: a full save writes nothing back into the deleted folder.
run save-all flush
expect Saved the game
run orevault vault folder 0000000a-0000-4000-8000-00000000000a
expect exists=false deleting=false far_region=false
# Check 3: a new level under the same key, from a fresh stem, under the other type.
run orevault vault create 0000000a-0000-4000-8000-00000000000a expanded
expect type=orevault:ore_vault_expanded$
log Registered Vault orevault:vault_0000000a00004000800000000000000a \(orevault:ore_vault_expanded,
run execute in orevault:vault_0000000a00004000800000000000000a run forceload add 0 0
expect Marked chunk
run execute in orevault:vault_0000000a00004000800000000000000a if block 0 -64 0 minecraft:bedrock
expect Test passed
# Check 5, headless form: a command in the key reaches the new level and breaks a block in it.
run execute in orevault:vault_0000000a00004000800000000000000a run orevault break 0 250 0
expect gone=true
run execute in orevault:vault_0000000a00004000800000000000000a if block 600 251 600 minecraft:gold_block
expect Test failed|That position is not loaded
log (?s)Registered Vault orevault:vault_0000000a00004000800000000000000a \(orevault:ore_vault_expanded,.*tick orevault:vault_0000000a00004000800000000000000a server_tick=
nolog (?s)Unloading Vault orevault:vault_0000000a00004000800000000000000a.*(/ERROR\]|Exception)
